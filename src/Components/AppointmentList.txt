import React, { useState } from "react";

const AppointmentList = ({
	appointments,
	deleteAppointment,
	editAppointment,
	clearAppointments,
}) => {
	const [editedIndex, setEditedIndex] = useState(null);
	const [editedName, setEditedName] = useState("");
	const [editedAge, setEditedAge] = useState("");
	const [editedAddress, setEditedAddress] = useState("");
	const [editedDate, setEditedDate] = useState("");
	const [editedTimeSlot, setEditedTimeSlot] = useState("");

	const handleEdit = (index) => {
		const appointment = appointments[index];
		setEditedIndex(index);
		setEditedName(appointment.name);
		setEditedAge(appointment.age);
		setEditedAddress(appointment.address);
		setEditedDate(appointment.date);
		setEditedTimeSlot(appointment.timeSlot);
	};

	const handleSaveEdit = (index) => {
		editAppointment(index, {
			name: editedName,
			age: editedAge,
			address: editedAddress,
			date: editedDate,
			timeSlot: editedTimeSlot,
		});
		setEditedIndex(null);
		setEditedName("");
		setEditedAge("");
		setEditedAddress("");
		setEditedDate("");
		setEditedTimeSlot("");
	};

	const handleCancelEdit = () => {
		setEditedIndex(null);
		setEditedName("");
		setEditedAge("");
		setEditedAddress("");
		setEditedDate("");
		setEditedTimeSlot("");
	};

	return (
		<div className="container">
			<h1>Appointment List</h1>
			<table id="list">
				<thead>
					<tr>
						<th>ID</th>
						<th>Name</th>
						<th>Age</th>
						<th>Address</th>
						<th>Date</th>
						<th>Time Slot</th>
						<th>Action</th>
					</tr>
				</thead>
				<tbody>
					{appointments.map((appointment, index) => (
						<tr key={index}>
							<td>{index + 1}</td>
							<td>
								{editedIndex === index ? (
									<input
										type="text"
										value={editedName}
										onChange={(e) =>
											setEditedName(e.target.value)
										}
									/>
								) : (
									appointment.name
								)}
							</td>
							<td>
								{editedIndex === index ? (
									<input
										type="number"
										value={editedAge}
										onChange={(e) =>
											setEditedAge(e.target.value)
										}
									/>
								) : (
									appointment.age
								)}
							</td>
							<td>
								{editedIndex === index ? (
									<input
										type="text"
										value={editedAddress}
										onChange={(e) =>
											setEditedAddress(e.target.value)
										}
									/>
								) : (
									appointment.address
								)}
							</td>
							<td>
								{editedIndex === index ? (
									<input
										type="date"
										value={editedDate}
										onChange={(e) =>
											setEditedDate(e.target.value)
										}
									/>
								) : (
									appointment.date
								)}
							</td>
							<td>
								{editedIndex === index ? (
									<input
										type="time"
										value={editedTimeSlot}
										onChange={(e) =>
											setEditedTimeSlot(e.target.value)
										}
									/>
								) : (
									appointment.timeSlot
								)}
							</td>
							<td>
								{editedIndex === index ? (
									<>
										<button
											className="save"
											onClick={() =>
												handleSaveEdit(index)
											}
										>
											Save
										</button>
										<button
											className="cancel"
											onClick={handleCancelEdit}
										>
											Cancel
										</button>
									</>
								) : (
									<>
										<button
											className="edit"
											onClick={() => handleEdit(index)}
										>
											Edit
										</button>
										<button
											className="delete"
											onClick={() =>
												deleteAppointment(index)
											}
										>
											Delete
										</button>
									</>
								)}
							</td>
						</tr>
					))}
				</tbody>
			</table>
			<button onClick={clearAppointments}>Clear All Appointments</button>
		</div>
	);
};

export default AppointmentList;
